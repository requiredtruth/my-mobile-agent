cmake_minimum_required(VERSION 3.22)
project(mobileagent_protocol LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
add_subdirectory(../vendor/llama.cpp llama)
add_executable(chat-format-test ChatFormatTest.cpp)
target_link_libraries(chat-format-test PRIVATE llama-common)
# Compile the JNI bridge on the host too, catching API drift before Android builds.
find_package(JNI REQUIRED)
add_library(bridge-check OBJECT ../app/src/main/cpp/bridge.cpp)
target_include_directories(bridge-check PRIVATE ${JNI_INCLUDE_DIRS})
target_link_libraries(bridge-check PRIVATE llama-common)
