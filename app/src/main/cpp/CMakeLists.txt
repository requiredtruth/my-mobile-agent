cmake_minimum_required(VERSION 3.22)
project(mobileagent LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
if(ANDROID AND CMAKE_ANDROID_ARCH_ABI STREQUAL "arm64-v8a")
    set(GGML_CPU_KLEIDIAI ON CACHE BOOL "" FORCE)
    set(GGML_VULKAN ON CACHE BOOL "" FORCE)
    find_library(android_vulkan vulkan REQUIRED)
    set(Vulkan_LIBRARY "${android_vulkan}" CACHE FILEPATH "" FORCE)
    set(Vulkan_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../../vendor/vulkan-headers/include" CACHE PATH "" FORCE)
    find_program(Vulkan_GLSLC_EXECUTABLE glslc REQUIRED NO_CMAKE_FIND_ROOT_PATH)
    set(SPIRV-Headers_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../../vendor/spirv-install/share/cmake/SPIRV-Headers" CACHE PATH "" FORCE)
endif()
add_subdirectory(../../../../vendor/llama.cpp llama)
if(TARGET ggml-vulkan)
    target_include_directories(ggml-vulkan PRIVATE
        "${CMAKE_CURRENT_SOURCE_DIR}/../../../../vendor/spirv-install/include")
endif()
add_library(mobileagent SHARED bridge.cpp)
target_link_libraries(mobileagent PRIVATE llama llama-common android log)
target_link_options(mobileagent PRIVATE "-Wl,-z,max-page-size=16384")
